SELECT
    ARTICLE_TITLE_CLEAN,
    PRIORITY_TIER,
    SIGNAL_SCORE,
    SIGNAL_REASONS,
    MATCHED_COMPANIES,
    MATCHED_COMPANY_CATEGORIES,
    ARTICLE_URL
FROM PHARMA_NEWS_SANDBOX.NEWS.V_PHARMA_NEWS_PRIORITY_TIER_V2
WHERE PRIORITY_TIER = 'MONITOR'
ORDER BY SIGNAL_SCORE DESC, RECEIVED_TS_PARSED DESC NULLS LAST
LIMIT 50;
