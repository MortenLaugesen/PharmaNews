-- ============================================================
-- 99 - Manual run
-- This file refreshes AI summaries and previews the email + quality checks.
-- The email send call is kept commented out.
-- ============================================================

CALL PHARMA_NEWS_SANDBOX.NEWS.SP_REFRESH_AI_SUMMARIES();


-- Preview email and quality checks
SELECT
    TO_EMAIL,
    EMAIL_SUBJECT,
    REGEXP_COUNT(EMAIL_BODY, '<b>Summary:</b>') AS SUMMARY_BLOCKS,
    REGEXP_COUNT(EMAIL_BODY, 'Source text was limited') AS FALLBACK_BLOCKS,
    LEFT(EMAIL_BODY, 30000) AS EMAIL_BODY_PREVIEW
FROM PHARMA_NEWS_SANDBOX.NEWS.V_ALTERYX_DAILY_PHARMA_NEWS_EMAIL;


-- Optional operational checks
SELECT
    COUNT(*) AS STAGING_ROWS
FROM PHARMA_NEWS_SANDBOX.NEWS.STG_PHARMA_NEWS_ARTICLES_V1;

SELECT
    PRIORITY_TIER,
    COUNT(*) AS CNT
FROM PHARMA_NEWS_SANDBOX.NEWS.V_PHARMA_NEWS_DIGEST_QUEUE_V2
WHERE RECEIVED_TS_PARSED >= DATEADD('day', -7, CURRENT_TIMESTAMP())
GROUP BY PRIORITY_TIER
ORDER BY CNT DESC;

SELECT
    DIGEST_RANK,
    ARTICLE_TITLE,
    PRIORITY_TIER,
    SIGNAL_SCORE,
    EMAIL_SOURCE_TYPE,
    PUBLISH_DATE,
    LEFT(ARTICLE_SUMMARY_INPUT, 500) AS SUMMARY_INPUT_PREVIEW
FROM PHARMA_NEWS_SANDBOX.NEWS.V_PHARMA_NEWS_DIGEST_QUEUE_V2
WHERE DIGEST_RANK <= 10
ORDER BY DIGEST_RANK;

SELECT
    ARTICLE_TITLE,
    EMAIL_SOURCE_TYPE,
    PUBLISH_DATE,
    LEFT(AI_SUMMARY, 1000) AS AI_SUMMARY_PREVIEW,
    CREATED_AT
FROM PHARMA_NEWS_SANDBOX.NEWS.PHARMA_NEWS_AI_SUMMARIES
WHERE SUMMARY_VERSION = 'v4_cost_effective_top10'
ORDER BY CREATED_AT DESC
LIMIT 10;
