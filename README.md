SELECT
    TO_EMAIL,
    EMAIL_SUBJECT,
    REGEXP_COUNT(EMAIL_BODY, '<b>Summary:</b>') AS SUMMARY_BLOCKS,
    REGEXP_COUNT(EMAIL_BODY, 'No AI summary available') AS MISSING_SUMMARY_BLOCKS,
    LEFT(EMAIL_BODY, 12000) AS EMAIL_BODY_PREVIEW
FROM PHARMA_NEWS_SANDBOX.NEWS.V_ALTERYX_DAILY_PHARMA_NEWS_EMAIL;
