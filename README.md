-- 3. Check om priority viewet overhovedet har rows
SELECT
    PRIORITY_TIER,
    COUNT(*) AS CNT
FROM PHARMA_NEWS_SANDBOX.NEWS.V_PHARMA_NEWS_PRIORITY_TIER_V2
GROUP BY PRIORITY_TIER
ORDER BY CNT DESC;

-- 4. Check digest queue uden dato-filter
SELECT
    PRIORITY_TIER,
    COUNT(*) AS CNT
FROM PHARMA_NEWS_SANDBOX.NEWS.V_PHARMA_NEWS_DIGEST_QUEUE_V2
GROUP BY PRIORITY_TIER
ORDER BY CNT DESC;

-- 5. Check digest queue med 30 dages filter
SELECT *
FROM PHARMA_NEWS_SANDBOX.NEWS.V_PHARMA_NEWS_DIGEST_QUEUE_V2
WHERE RECEIVED_TS_PARSED >= DATEADD('day', -30, CURRENT_TIMESTAMP())
ORDER BY RECEIVED_TS_PARSED DESC NULLS LAST
LIMIT 50;
