SELECT
    D.DIGEST_RANK,
    D.ARTICLE_TITLE,
    D.ARTICLE_URL,
    D.STORY_KEY,
    D.SIGNAL_SCORE,
    D.PRIORITY_TIER,
    D.DEEP_DIVE_FLAG
FROM PHARMA_NEWS_SANDBOX.NEWS.V_PHARMA_NEWS_DIGEST_QUEUE_V2 D
LEFT JOIN PHARMA_NEWS_SANDBOX.NEWS.PHARMA_NEWS_ENRICHED_ARTICLES E
    ON D.ARTICLE_URL = E.ARTICLE_URL
WHERE D.DEEP_DIVE_FLAG = TRUE
  AND D.ARTICLE_URL IS NOT NULL
  AND E.ARTICLE_URL IS NULL
ORDER BY D.DIGEST_RANK;
